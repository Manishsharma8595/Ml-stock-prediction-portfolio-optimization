class EqualWeightPortfolio(QCAlgorithm):
    """
    Q2(i): Equal-weighted, long-only, buy-and-hold portfolio of 10 US stocks.
    Buy once at the start of the period, never trade again, compare with NYSE Composite (NYA).
    """

    def initialize(self):
        # Set the backtest dates and starting capital according to the assignment requirements.
        self.set_start_date(2023, 1, 1)   # The requested period starts on 1 January 2023. Since this date is a Sunday, the first purchase can take place when the market next opens, on 3 January.
        self.set_end_date(2026, 1, 1)
        self.set_cash(100000)             # Start the portfolio with $100,000 in cash.

        # Store the ten stock tickers specified in the assignment. These are the only individual stocks included in this portfolio.
        tickers = ["AAPL", "MSFT", "JPM", "LLY", "TGT", "AOS", "NOG", "POR", "FDX", "PYPL"]

        # Minute-level data allows the algorithm to place the initial purchases shortly after the market opens on 3 January 2023.
        # With daily resolution, an order generated from the first daily bar may be filled at the next available daily opening, potentially delaying the intended purchase.
        self.stocks = [self.add_equity(t, Resolution.MINUTE).symbol for t in tickers]

        # Add the NYSE Composite index so portfolio performance can be compared with a broader market benchmark.
        nya = self.add_index("NYA", Resolution.DAILY).symbol
        # The benchmark is used in QuantConnect performance charts and in benchmark-relative statistics,
        # including measures such as alpha, beta, information ratio, and tracking error.
        self.set_benchmark(nya)

        # This flag records whether the initial portfolio allocation has already been made.
        self.bought = False

    def on_data(self, data: Slice):
        # Once the stocks have been purchased, no further orders are placed. This keeps the strategy as a buy-and-hold portfolio.
        if self.bought:
            return

        # Check that every stock has a valid price before submitting the portfolio orders. Some securities may not have a price available in the first minute,
        # so the algorithm waits until price data is available for all ten holdings.
        if not all(self.securities[s].price > 0 for s in self.stocks):
            return

        # Divide the portfolio equally across the ten stocks, assigning a target weight of 10% to each one.
        # QuantConnect may retain a small amount of cash to help cover fees and order costs, so the actual amount invested in each stock can be slightly below $10,000.
        # As a result, the holdings may not use the full $100,000 at the moment of purchase.
        weight = 1.0 / len(self.stocks)
        self.set_holdings([PortfolioTarget(s, weight) for s in self.stocks])

        self.bought = True
        self.log(f"Bought all 10 stocks at {self.time}")

    def on_end_of_algorithm(self):
        # At the end of the backtest, print the final portfolio value and each stock holding for use in the results discussion.
        self.log(f"Final portfolio value: {self.portfolio.total_portfolio_value:,.2f}")
        for s in self.stocks:
            h = self.portfolio[s]
            self.log(f"{s.value}: shares={h.quantity}, value={h.holdings_value:,.2f}, "
                     f"profit={h.unrealized_profit:,.2f}")
