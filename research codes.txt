# CELL 5: Calculate price-based performance and risk measures for each stock over the sample period.
import numpy as np

trading_days = 252     # Approximate number of trading days in a year, used to annualise daily measures.
risk_free = 0.04       # Assumed annual risk-free rate of 4%; this assumption should also be stated in the written report.

# SPY is used as a broad-market proxy when calculating each stock's beta.
spy_symbol = qb.add_equity("SPY", Resolution.DAILY).symbol
spy_hist = qb.history(spy_symbol, datetime(2023,1,1), datetime(2026,1,1), Resolution.DAILY)
spy_returns = spy_hist["close"].droplevel(0).pct_change().dropna()

# Create a results table with one row for each stock and add the selected measures as columns.
stats = pd.DataFrame(index=returns.columns)

# Total return measures the percentage change between the first and last observed closing prices.
stats["Total return (%)"] = (close.iloc[-1] / close.iloc[0] - 1) * 100

# Annualised volatility scales the standard deviation of daily returns to an approximate yearly measure of risk.
stats["Annual volatility (%)"] = returns.std() * np.sqrt(trading_days) * 100

# The Sharpe ratio compares annualised excess return with annualised volatility; a higher value indicates more return per unit of risk.
annual_return = returns.mean() * trading_days
stats["Sharpe ratio"] = (annual_return - risk_free) / (returns.std() * np.sqrt(trading_days))

# Maximum drawdown measures the largest percentage decline from a previous cumulative peak to a later trough.
growth = (1 + returns).cumprod()
stats["Max drawdown (%)"] = (growth / growth.cummax() - 1).min() * 100

# Beta measures the stock's sensitivity to movements in SPY; a value near 1 indicates similar market sensitivity.
stats["Beta vs SPY"] = [returns[t].cov(spy_returns) / spy_returns.var() for t in returns.columns]

# The 5th percentile of daily returns is used as the 95% daily VaR threshold, showing the return level in the lower tail of outcomes.
stats["Daily VaR 95% (%)"] = returns.quantile(0.05) * 100

stats.round(2)

def safe(getter):
    """Try to read a value. If it is missing, return NaN instead of crashing."""
    try:
        return float(getter())
    except Exception:
        return float("nan")

        "ROE (%)":            safe(lambda: f.operation_ratios.roe.value) * 100,
    })

# CELL 8: Prepare the input features and next-day closing-price target for the machine-learning models.
import numpy as np
import pandas as pd

chosen = ["JPM", "PYPL"]          # JPM and PYPL are the two securities selected for the prediction exercise.

# Download daily open, high, low, close, and volume observations for the two selected securities.
hist = qb.history([symbols[t] for t in chosen],
                  datetime(2023, 1, 1), datetime(2026, 1, 1), Resolution.DAILY)

def make_features(df):
    """Takes one stock's table and adds input columns (features)."""
    d = pd.DataFrame(index=df.index)
    d["close"]   = df["close"]
    d["ret_1"]   = df["close"].pct_change(1)      # Percentage change in closing price over one trading day.
    d["ret_5"]   = df["close"].pct_change(5)      # Percentage change in closing price over the previous five trading days.
    d["sma_5"]   = df["close"].rolling(5).mean()  # Average closing price over the latest five observations.
    d["sma_20"]  = df["close"].rolling(20).mean() # Average closing price over the latest twenty observations.
    d["vol_chg"] = df["volume"].pct_change()      # Percentage change in trading volume compared with the previous observation.
    d["volume"]  = df["volume"]

# RSI is a 14-period momentum measure intended to indicate the balance between recent upward and downward price movements.
    delta = df["close"].diff()
    gain = delta.clip(lower=0).rolling(14).mean()
    loss = (-delta.clip(upper=0)).rolling(14).mean()
    d["rsi_14"] = 100 - 100 / (1 + gain / loss)

# The target is shifted one row upward so each row's features are paired with the following trading day's closing price.
    d["target"] = df["close"].shift(-1)
    return d.dropna()

data = {}
for t in chosen:                                   # Process each selected stock separately, applying the same feature construction steps to both.
    df = hist.loc[symbols[t]]                      # Select the historical observations belonging to the current ticker before creating its features.
    data[t] = make_features(df)

data["JPM"].head()

# CELL 9: Train an XGBoost regression model to estimate the next trading day's closing price.
from xgboost import XGBRegressor
from sklearn.metrics import mean_squared_error, mean_absolute_percentage_error
import matplotlib.pyplot as plt

feature_cols = ["close","ret_1","ret_5","sma_5","sma_20","vol_chg","volume","rsi_14"]
xgb_results = {}

for t in chosen:
    d = data[t]
    split = int(len(d) * 0.7)                     # Use the first 70% of observations for model fitting, leaving the later 30% for out-of-sample evaluation.
    train, test = d.iloc[:split], d.iloc[split:]  # Keep the time order intact: earlier observations train the model and later observations test its predictions.

    model = XGBRegressor(n_estimators=300, learning_rate=0.05, max_depth=4)
    model.fit(train[feature_cols], train["target"])

    pred = model.predict(test[feature_cols])
    rmse = np.sqrt(mean_squared_error(test["target"], pred))
    mape = mean_absolute_percentage_error(test["target"], pred) * 100
    xgb_results[t] = (rmse, mape)

    plt.figure(figsize=(12, 5))
    plt.plot(test.index, test["target"], label="Actual close")
    plt.plot(test.index, pred, label="XGBoost predicted close")
    plt.title(f"{t}: XGBoost, actual vs predicted next-day close")
    plt.legend(); plt.show()
    print(t, "RMSE:", round(rmse, 2), " MAPE (%):", round(mape, 2))

# CELL 10: Train an LSTM neural network for next-day closing-price prediction.
from sklearn.preprocessing import MinMaxScaler
import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense, Dropout

tf.random.set_seed(42); np.random.seed(42)   # Set random seeds to make the model's results more reproducible across runs.

window = 10                                  # Each LSTM input sequence contains the most recent ten days of feature values.
lstm_results = {}

for t in chosen:
    d = data[t]
    split = int(len(d) * 0.7)                # Use the same chronological 70% training and 30% testing split as the XGBoost model for comparison.

# Scale the input features and target to the 0-to-1 range, which can help neural-network training.
# Fit each scaler using training observations only, avoiding the use of test-period information when estimating scaling limits.
    x_scaler = MinMaxScaler().fit(d[feature_cols].iloc[:split])
    y_scaler = MinMaxScaler().fit(d[["target"]].iloc[:split])
    X_all = x_scaler.transform(d[feature_cols])
    y_all = y_scaler.transform(d[["target"]])

# Form rolling sequences in which ten consecutive days of features are used to predict the target associated with the sequence's final day.
    X_seq, y_seq = [], []
    for i in range(window - 1, len(d)):
        X_seq.append(X_all[i - window + 1 : i + 1])
        y_seq.append(y_all[i])
    X_seq, y_seq = np.array(X_seq), np.array(y_seq)

    n_train = split - window + 1             # Calculate how many sequences belong to the training period after accounting for the ten-day input window.
    X_train, y_train = X_seq[:n_train], y_seq[:n_train]
    X_test = X_seq[n_train:]
    actual = d["target"].iloc[split:]        # Keep the actual target values for the test period so predictions can be evaluated against observed closing prices.

# The network contains an LSTM layer to learn sequential patterns, dropout to help limit overfitting, and one output for the predicted price.
    model = Sequential([
        LSTM(32, input_shape=(window, len(feature_cols))),
        Dropout(0.2),
        Dense(1)
    ])
    model.compile(optimizer="adam", loss="mse")
    model.fit(X_train, y_train, epochs=50, batch_size=16, verbose=0)

# Convert the model's scaled predictions back to the original dollar-price scale before calculating error measures.
    pred = y_scaler.inverse_transform(model.predict(X_test, verbose=0)).ravel()

    rmse = np.sqrt(mean_squared_error(actual, pred))
    mape = mean_absolute_percentage_error(actual, pred) * 100
    lstm_results[t] = (rmse, mape)

    plt.figure(figsize=(12, 5))
    plt.plot(actual.index, actual, label="Actual close")
    plt.plot(actual.index, pred, label="LSTM predicted close")
    plt.title(f"{t}: LSTM, actual vs predicted next-day close")
    plt.legend(); plt.show()
    print(t, "RMSE:", round(rmse, 2), " MAPE (%):", round(mape, 2))

# CELL 15 (research notebook): Question 2(i) — compare the equal-weight portfolio with the NYSE Composite index.
# The comparison uses the holdings recorded in the portfolio backtest and index data for the same period.
#
# Run the earlier research cells first so that the closing-price table named close has already been created.
# =============================================================================
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# 1. Record the number of shares purchased for each stock at the start of the portfolio backtest, based on the Orders tab.
shares = {"LLY": 28, "PYPL": 137, "JPM": 81, "NOG": 406, "AOS": 185,
          "POR": 239, "AAPL": 78, "FDX": 94, "MSFT": 42, "TGT": 75}

# The backtest purchases whole shares, so a small amount of the initial cash remains uninvested after transaction costs.
# This leftover cash is taken from the backtest log: ending portfolio value less the ending market value of the ten stock positions.
cash_left = 512.26
start_cash = 100000.0
risk_free = 0.04          # Use the same assumed annual risk-free rate as in the earlier performance-statistics section.
trading_days = 252

# 2. Calculate the portfolio's daily market value by multiplying each holding by its closing price and adding the remaining cash.
prices = close[list(shares.keys())].copy()                # 'close' table from Cell 1
prices.index = pd.to_datetime(prices.index).normalize()   # keep only the date, drop the time
portfolio = (prices * pd.Series(shares)).sum(axis=1) + cash_left

# 3. Retrieve daily NYSE Composite index data to use as the benchmark for the portfolio.
nya = qb.add_index("NYA", Resolution.DAILY).symbol
nya_hist = qb.history(nya, datetime(2023, 1, 1), datetime(2026, 1, 1), Resolution.DAILY)
nya_hist = nya_hist.droplevel(0)                          # keep only the date as the index
nya_hist.index = pd.to_datetime(nya_hist.index).normalize()

# 4. Align the portfolio and index series on matching dates so their performance is compared over a common sample.
both = pd.concat([portfolio.rename("Portfolio"), nya_hist["close"].rename("NYA")],
                 axis=1).dropna()

# Add a starting observation before the first common trading date, with the portfolio at its initial $100,000 value and the index at its opening level.
# The backtest entered the stock positions shortly after the market opened on 3 January 2023, so the index opening level is used as the benchmark starting point.
start_row = pd.DataFrame({"Portfolio": [start_cash], "NYA": [float(nya_hist["open"].iloc[0])]},
                         index=[both.index[0] - pd.Timedelta(days=1)])
levels = pd.concat([start_row, both])

print("Portfolio value on the last day in this notebook:", round(levels["Portfolio"].iloc[-1], 2),
      " (backtest: 161,361.95)")

# 5. Calculate daily percentage returns from the aligned portfolio and index value series.
daily = levels.pct_change().dropna()

# 6. Calculate a consistent set of performance and risk measures for each value series.
def summary(level):
    r = level.pct_change().dropna()
    years = (level.index[-1] - level.index[0]).days / 365.25
    vol = r.std() * np.sqrt(trading_days)
    return {
        "Total return (%)":        (level.iloc[-1] / level.iloc[0] - 1) * 100,
        "Annualised return (%)":   ((level.iloc[-1] / level.iloc[0]) ** (1 / years) - 1) * 100,
        "Average daily return (%)": r.mean() * 100,
        "Daily volatility (%)":    r.std() * 100,
        "Annual volatility (%)":   vol * 100,
        "Sharpe ratio":            (r.mean() * trading_days - risk_free) / vol,
        "Max drawdown (%)":        (level / level.cummax() - 1).min() * 100,
        "Best day (%)":            r.max() * 100,
        "Worst day (%)":           r.min() * 100,
    }

table = pd.DataFrame({"Equal-weight portfolio": summary(levels["Portfolio"]),
                      "NYSE Composite (NYA)":   summary(levels["NYA"])})
print(table.round(2))

# Calculate additional comparisons to show the portfolio's relationship with the benchmark and how often it outperformed on a daily basis.
beta = daily["Portfolio"].cov(daily["NYA"]) / daily["NYA"].var()
corr = daily["Portfolio"].corr(daily["NYA"])
beat = (daily["Portfolio"] > daily["NYA"]).mean() * 100
print("\nBeta of portfolio vs NYA:", round(beta, 2))
print("Correlation of daily returns:", round(corr, 2))
print("% of days the portfolio did better than NYA:", round(beat, 1))
print("Portfolio end value ($):", round(levels["Portfolio"].iloc[-1], 2))
print("$100,000 in NYA would have become ($):",
      round(start_cash * levels["NYA"].iloc[-1] / levels["NYA"].iloc[0], 2))
